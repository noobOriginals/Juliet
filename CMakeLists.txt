cmake_minimum_required(VERSION 3.26.0)

if (APPLE)
    project(Juliet VERSION 0.1.0 LANGUAGES C CXX OBJC OBJCXX)
elseif(WIN32)
    project(Juliet VERSION 0.1.0 LANGUAGES C CXX CUDA)
else()
    project(Juliet VERSION 0.1.0 LANGUAGES C CXX)
endif()

set(FullOutputDir "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${FullOutputDir}/static-libs")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${FullOutputDir}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${FullOutputDir}")

set(SrcPath "${CMAKE_SOURCE_DIR}/src")
set(HeaderPath "${CMAKE_SOURCE_DIR}/include")
if (APPLE)
    file(GLOB_RECURSE Sources CONFIGURE_DEPENDS "${SrcPath}/*.c" "${SrcPath}/*.cpp" "${SrcPath}/*.m" "${SrcPath}/*.mm")
elseif(WIN32)
    file(GLOB_RECURSE Sources CONFIGURE_DEPENDS "${SrcPath}/*.c" "${SrcPath}/*.cpp" "${SrcPath}/*.cu")
else()
    file(GLOB_RECURSE Sources CONFIGURE_DEPENDS "${SrcPath}/*.c" "${SrcPath}/*.cpp")
endif()

include_directories(${HeaderPath} $ENV{OPENGL_PATH})

find_package(Threads REQUIRED)

add_executable(${PROJECT_NAME} ${Sources})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
if (WIN32)
    set_target_properties(${PROJECT_NAME} PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
target_compile_options(${PROJECT_NAME} PRIVATE -O3)
if (APPLE)
    target_link_libraries(${PROJECT_NAME} PRIVATE Threads::Threads "-framework Foundation" "-framework Metal")
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE Threads::Threads)
endif()
